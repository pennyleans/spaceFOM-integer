cmake_minimum_required(VERSION 3.20)
project(OrbitalSpaceFom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED OPENRTI_ROOT)
  message(FATAL_ERROR "OPENRTI_ROOT must name the separately installed RTI")
endif()
foreach(role publisher observer)
  add_executable(spacefom-${role} ${role}.cpp)
  target_include_directories(spacefom-${role} SYSTEM PRIVATE "${OPENRTI_ROOT}/include/rti1516e")
  target_link_libraries(spacefom-${role} PRIVATE "${OPENRTI_ROOT}/lib/rti1516e.lib" "${OPENRTI_ROOT}/lib/fedtime1516e.lib")
  target_compile_definitions(spacefom-${role} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(spacefom-${role} PRIVATE /W4 /WX /EHsc /external:W0)
  else()
    target_compile_options(spacefom-${role} PRIVATE -Wall -Wextra -Wshadow -Werror)
  endif()
endforeach()
