cmake_minimum_required(VERSION 3.20)
project(SpaceFomStateRoundTrip LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED OPENRTI_ROOT)
  message(FATAL_ERROR "OPENRTI_ROOT must name the separately installed RTI")
endif()
add_executable(spacefom-state-roundtrip state-roundtrip.cpp)
target_include_directories(spacefom-state-roundtrip SYSTEM PRIVATE "${OPENRTI_ROOT}/include/rti1516e")
target_link_libraries(spacefom-state-roundtrip PRIVATE "${OPENRTI_ROOT}/lib/rti1516e.lib" "${OPENRTI_ROOT}/lib/fedtime1516e.lib")
if(MSVC)
  target_compile_options(spacefom-state-roundtrip PRIVATE /W4 /WX /EHsc /external:W0)
else()
  target_compile_options(spacefom-state-roundtrip PRIVATE -Wall -Wextra -Wshadow -Werror)
endif()
